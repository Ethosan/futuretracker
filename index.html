Trade Journal
Setup guide
A personal futures trading journal — MES, MNQ, MGC, XAUUSD. Tracks Live and Prop accounts separately. Syncs to your own Google Sheet. Follow the steps below once per device.
1
Create a Google Sheet
Go to sheets.new → name it anything (e.g. "Trade Journal"). The journal will set everything else up automatically.
2
Add the Apps Script
Inside your Google Sheet → Extensions → Apps Script → delete all existing code → paste this:
function doGet(e) { var cb = e.parameter.callback; var payload = e.parameter.payload; var ss = SpreadsheetApp.getActiveSpreadsheet(); // find or create the Trades sheet var sheet = ss.getSheetByName('Trades'); if (!sheet) { var s = ss.getSheets()[0]; s.setName('Trades'); sheet = s; } if (payload) { var data = JSON.parse(decodeURIComponent(payload)); if (data.action === 'setup') { sheet.clearContents(); sheet.appendRow(data.headers); var r = sheet.getRange(1, 1, 1, data.headers.length); r.setBackground('#2a2a2a'); r.setFontColor('#c8f56a'); r.setFontWeight('bold'); r.setFontFamily('DM Mono'); sheet.setFrozenRows(1); sheet.setColumnWidth(1, 180); for (var i = 2; i <= data.headers.length; i++) { sheet.setColumnWidth(i, 140); } return out(cb, { status: 'ok' }); } if (data.action === 'append') { sheet.appendRow(data.row); return out(cb, { status: 'ok' }); } if (data.action === 'update') { var id = data.row[0]; var vals = sheet.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (String(vals[i][0]) === String(id)) { sheet.getRange(i + 1, 1, 1, data.row.length).setValues([data.row]); return out(cb, { status: 'ok' }); } } return out(cb, { status: 'not_found' }); } } var values = sheet.getDataRange().getDisplayValues(); return out(cb, { values: values }); } function out(cb, data) { return ContentService .createTextOutput(cb + '(' + JSON.stringify(data) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); }
3
Deploy as a Web App
1. Press Ctrl+S to save
2. Click Deploy (top right) → New deployment
3. Click the gear icon ⚙ next to "Select type" → choose Web app
4. Under "Execute as" → select Me
5. Under "Who has access" → select Anyone
6. Click Deploy → click Authorize access → sign in with your Google account → allow
7. Copy the Web app URL shown — it looks like:
https://script.google.com/macros/s/ABC.../exec

⚠ Every time you edit the script you must create a New deployment — not "Manage deployments" — otherwise changes won't take effect.
